Private Sub EnsureDayCellClass(ByVal vbProj As Object)
     Dim cls As Object
     On Error Resume Next
     Set cls = vbProj.VBComponents("clsDayCell")
     On Error GoTo 0
    If cls Is Nothing Then
        Set cls = vbProj.VBComponents.Add(2)
        cls.Name = "clsDayCell"
    Else
        With cls.CodeModule
            .DeleteLines 1, .CountOfLines
        End With
    End If
    cls.CodeModule.AddFromString _
        "Option Explicit" & vbCrLf & _
        "Public WithEvents Cell As MSForms.Label" & vbCrLf & _
        "Private mHost As frmNames" & vbCrLf & _
        "Public Sub Init(ctrl As MSForms.Label, host As frmNames)" & vbCrLf & _
        "    Set Cell = ctrl" & vbCrLf & _
        "    Set mHost = host" & vbCrLf & _
        "End Sub" & vbCrLf & _
        "Private Sub Cell_Click()" & vbCrLf & _
        "    mHost.HandleDayClick Cell" & vbCrLf & _
        "End Sub" & vbCrLf & _
        "Private Sub Cell_DblClick(ByVal Cancel As MSForms.ReturnBoolean)" & vbCrLf & _
        "    mHost.HandleDayDoubleClick Cell" & vbCrLf & _
        "End Sub"
 End Sub

Public Sub ShowNamesForm()
    frmNames.Show vbModeless
End Sub

Private Function ReadFileToString(ByVal path As String) As String
    Dim fso As Object
    Dim ts As Object
    Set fso = CreateObject("Scripting.FileSystemObject")
    Set ts = fso.OpenTextFile(path, 1, False)
    ReadFileToString = ts.ReadAll
    ts.Close
End Function
