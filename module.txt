Option Explicit

Public Sub Auto_Open()
    On Error GoTo AutoOpenFail
    ShowRankCalendarForm
    Exit Sub
AutoOpenFail:
    MsgBox "frmRankCalendar başlatılamadı. Hata: " & Err.Number & " - " & Err.Description, vbCritical
End Sub

Public Sub RebuildRankCalendarForm()
    BuildRankCalendarForm
    ShowRankCalendarForm
End Sub

Public Sub ShowRankCalendarForm()
    If Not FormExists("frmRankCalendar") Then
        BuildRankCalendarForm
    End If
    VBA.UserForms.Add("frmRankCalendar").Show vbModeless
End Sub

Private Sub EnsureDayCellClass(ByVal vbProj As Object)
    Dim cls As Object
    On Error Resume Next
    Set cls = vbProj.VBComponents("clsDayCell")
    On Error GoTo 0
    If cls Is Nothing Then
        Set cls = vbProj.VBComponents.Add(2)
        cls.Name = "clsDayCell"
    Else
        With cls.CodeModule
            .DeleteLines 1, .CountOfLines
        End With
    End If
    cls.CodeModule.AddFromString _
        "Option Explicit" & vbCrLf & _
        "Public WithEvents Cell As MSForms.Label" & vbCrLf & _
        "Private mHost As frmRankCalendar" & vbCrLf & _
        "Public Sub Init(ctrl As MSForms.Label, host As frmRankCalendar)" & vbCrLf & _
        "    Set Cell = ctrl" & vbCrLf & _
        "    Set mHost = host" & vbCrLf & _
        "End Sub" & vbCrLf & _
        "Private Sub Cell_Click()" & vbCrLf & _
        "    mHost.HandleDayClick Cell" & vbCrLf & _
        "End Sub" & vbCrLf & _
        "Private Sub Cell_DblClick(ByVal Cancel As MSForms.ReturnBoolean)" & vbCrLf & _
        "    mHost.HandleDayDoubleClick Cell" & vbCrLf & _
        "End Sub"
End Sub

Private Function FormExists(ByVal formName As String) As Boolean
    Dim cmp As Object
    For Each cmp In ThisWorkbook.VBProject.VBComponents
        If cmp.Type = 3 And cmp.Name = formName Then
            FormExists = True
            Exit Function
        End If
    Next cmp
End Function

Private Function GetRankCalendarFormCode() As String
    Dim lines As Collection
    Dim arr() As String
    Dim i As Long
    Set lines = New Collection
    lines.Add "Option Explicit"
    lines.Add ""
    lines.Add "Private Const SHEET_NAME As String = ""Sayfa1"""
    lines.Add "Private Const RANK_COLUMN As Long = 1        ' A"
    lines.Add "Private Const NAME_COLUMN As Long = 2        ' B"

    lines.Add "Private Const NAME_START_ROW As Long = 1"
    lines.Add "Private Const DATE_COLUMN As Long = 3        ' C for calendar dates"

    lines.Add "Private mDayCells(1 To 42) As clsDayCell"
    lines.Add "Private mCurrentMonth As Date"

    lines.Add "Private mSuppressMonthYear As Boolean"
    lines.Add ""
    lines.Add "Private Sub UserForm_Initialize()"

    lines.Add "    RenderMonth Date"
    lines.Add "    lblSelected.Caption = ""Seçim Yok"""
    lines.Add "    lblStatus.Caption = """""
    lines.Add "End Sub"
    lines.Add ""
    lines.Add "Private Sub LoadNameTable()"
    lines.Add "    Dim ws As Worksheet"
    lines.Add "    Dim lastRow As Long"
    lines.Add "    Dim i As Long"
    lines.Add "    lstEntries.Clear"
    lines.Add "    Set ws = ThisWorkbook.Worksheets(SHEET_NAME)"
    lines.Add "    lastRow = ws.Cells(ws.Rows.Count, RANK_COLUMN).End(xlUp).Row"
    lines.Add "    If lastRow < RANK_START_ROW Then Exit Sub"
    lines.Add "    For i = RANK_START_ROW To lastRow"
    lines.Add "        If Trim$(CStr(ws.Cells(i, RANK_COLUMN).Value)) <> """" And _"
    lines.Add "           Trim$(CStr(ws.Cells(i, NAME_COLUMN).Value)) <> """" Then"
    lines.Add "            lstEntries.AddItem ws.Cells(i, RANK_COLUMN).Value & "" | "" & ws.Cells(i, NAME_COLUMN).Value"

    lines.Add "End Sub"
    lines.Add ""
    lines.Add "Private Sub btnReloadList_Click()"
    lines.Add "    LoadNameTable"
    lines.Add "    ShowStatus ""Liste yenilendi."", False"
    lines.Add "End Sub"
    lines.Add ""

    lines.Add "    Dim selections As Variant"
    lines.Add "    Dim target As Range"
    lines.Add "    If Not PrepareSelections(selections) Then Exit Sub"

    lines.Add "        ShowStatus ""Önce çalışma sayfasında hedef hücre seçin."", True"

    lines.Add "    End If"
    lines.Add "    Set target = ActiveCell"
    lines.Add "    target.Value = Join(selections, vbLf)"
    lines.Add "    target.WrapText = True"
    lines.Add "    ShowStatus ""Seçimler aynı hücrede alt alta yazıldı."", False"
    lines.Add "    ClearSelections"
    lines.Add "End Sub"
    lines.Add ""
    lines.Add "Private Sub btnPlaceSeparateCells_Click()"
    lines.Add "    Dim selections As Variant"
    lines.Add "    Dim target As Range"
    lines.Add "    Dim i As Long"
    lines.Add "    If Not PrepareSelections(selections) Then Exit Sub"
    lines.Add "    If ActiveCell Is Nothing Then"
    lines.Add "        ShowStatus ""Önce çalışma sayfasında hedef hücre seçin."", True"
    lines.Add "        Exit Sub"
    lines.Add "    End If"
    lines.Add "    Set target = ActiveCell"
    lines.Add "    For i = LBound(selections) To UBound(selections)"
    lines.Add "        target.Offset(i, 0).Value = selections(i)"
    lines.Add "    Next i"
    lines.Add "    ShowStatus ""Seçimler alt alta hücrelere yazıldı."", False"
    lines.Add "    ClearSelections"
    lines.Add "End Sub"
    lines.Add ""
    lines.Add "Private Function PrepareSelections(ByRef combined As Variant) As Boolean"

    lines.Add "    Dim i As Long"
    lines.Add "    Dim raw As String"

    lines.Add "        ShowStatus ""En az bir satır seçin."", True"

    lines.Add "    ReDim combined(0 To chosen.Count - 1)"
    lines.Add "    For i = 1 To chosen.Count"
    lines.Add "        raw = chosen(i)"
    lines.Add "        combined(i - 1) = Trim$(Split(raw, ""|"")(0)) & "" "" & Trim$(Split(raw, ""|"")(1))"
    lines.Add "    Next i"
    lines.Add "    PrepareSelections = True"
    lines.Add "End Function"
    lines.Add ""
    lines.Add "Private Function GetSelectedEntries() As Collection"
    lines.Add "    Dim col As New Collection"
    lines.Add "    Dim i As Long"

    lines.Add "        If lstEntries.Selected(i) Then col.Add lstEntries.List(i)"
    lines.Add "    Next i"

    lines.Add "Private Sub InitialiseMonthYearSelectors(ByVal sourceDate As Date)"
    lines.Add "    Dim months As Variant"
    lines.Add "    Dim yr As Long"
    lines.Add "    months = Array(""Ocak"", ""Şubat"", ""Mart"", ""Nisan"", ""Mayıs"", ""Haziran"", _"
    lines.Add "                    ""Temmuz"", ""Ağustos"", ""Eylül"", ""Ekim"", ""Kasım"", ""Aralık"")"
    lines.Add "    cboMonth.List = months"
    lines.Add "    cboYear.Clear"
    lines.Add "    For yr = Year(sourceDate) To Year(sourceDate) + 10"
    lines.Add "        cboYear.AddItem CStr(yr)"
    lines.Add "    Next yr"
    lines.Add "    mSuppressMonthYear = True"
    lines.Add "    cboMonth.ListIndex = Month(sourceDate) - 1"
    lines.Add "    cboYear.Value = CStr(Year(sourceDate))"

    lines.Add "        Set lbl = fraCalendar.Controls.Add(""Forms.Label.1"", ""lblHead"" & c, True)"
    lines.Add "        With lbl"
    lines.Add "            .Caption = Format(DateSerial(2024, 1, 1 + c), ""ddd"")"
    lines.Add "            .Left = baseLeft + c * 28"
    lines.Add "            .Top = 12"
    lines.Add "            .Width = 28"
    lines.Add "            .Height = 18"
    lines.Add "            .BackColor = RGB(74, 60, 155)"
    lines.Add "            .ForeColor = vbWhite"
    lines.Add "            .Font.Size = 8"
    lines.Add "            .Font.Bold = True"
    lines.Add "            .TextAlign = fmTextAlignCenter"

    lines.Add "    Next c"
    lines.Add "    For i = 1 To 42"

    lines.Add "            .Left = baseLeft + ((i - 1) Mod 7) * 28"
    lines.Add "            .Top = baseTop + ((i - 1) \ 7) * rowSpacing"
    lines.Add "            .BackColor = RGB(245, 245, 250)"
    lines.Add "            .ForeColor = RGB(50, 50, 50)"
    lines.Add "            .BorderStyle = fmBorderStyleSingle"
    lines.Add "            .TextAlign = fmTextAlignCenter"
    lines.Add "            .Font.Size = 9"
    lines.Add "        End With"
    lines.Add "        Set mDayCells(i) = New clsDayCell"
    lines.Add "        mDayCells(i).Init lbl, Me"

    lines.Add "End Sub"

    lines.Add "Private Sub cboMonth_Change()"
    lines.Add "    If mSuppressMonthYear Then Exit Sub"
    lines.Add "    If cboMonth.ListIndex < 0 Then Exit Sub"
    lines.Add "    If Len(cboYear.Value) = 0 Then Exit Sub"

    lines.Add "End Sub"
    lines.Add ""
    lines.Add "Private Sub cboYear_Change()"
    lines.Add "    If mSuppressMonthYear Then Exit Sub"
    lines.Add "    If cboMonth.ListIndex < 0 Then Exit Sub"
    lines.Add "    If Not IsNumeric(cboYear.Value) Then Exit Sub"
    lines.Add "    RenderMonth DateSerial(CLng(cboYear.Value), cboMonth.ListIndex + 1, 1)"

    lines.Add "    mSuppressMonthYear = False"
    lines.Add "    lblMonth.Caption = UCase$(Format(mCurrentMonth, ""mmmm yyyy""))"

    lines.Add "        With mDayCells(i).Cell"

    lines.Add "                .BackColor = RGB(74, 60, 155)"

    lines.Add "Private Function ReadColumnValues(ByVal columnIndex As Long, ByVal startRow As Long) As Variant"
    lines.Add "    Dim ws As Worksheet"
    lines.Add "    Dim lastRow As Long"
    lines.Add "    Dim rng As Range"
    lines.Add "    Dim data As Variant"
    lines.Add "    Dim results As Collection"
    lines.Add "    Dim i As Long"
    lines.Add "    Set ws = ThisWorkbook.Worksheets(SHEET_NAME)"
    lines.Add "    lastRow = ws.Cells(ws.Rows.Count, columnIndex).End(xlUp).Row"
    lines.Add "    If lastRow < startRow Then Exit Function"
    lines.Add "    Set rng = ws.Range(ws.Cells(startRow, columnIndex), ws.Cells(lastRow, columnIndex))"
    lines.Add "    If rng.Cells.Count = 1 Then"
    lines.Add "        If Trim$(CStr(rng.Value)) = """" Then Exit Function"
    lines.Add "        ReadColumnValues = Array(Trim$(CStr(rng.Value)))"
    lines.Add "        Exit Function"
    lines.Add "    End If"
    lines.Add "    data = rng.Value"
    lines.Add "    Set results = New Collection"
    lines.Add "    For i = 1 To UBound(data, 1)"
    lines.Add "        If Trim$(CStr(data(i, 1))) <> """" Then results.Add Trim$(CStr(data(i, 1)))"
    lines.Add "    Next i"
    lines.Add "    If results.Count = 0 Then Exit Function"
    lines.Add "    Dim arr() As String"
    lines.Add "    ReDim arr(0 To results.Count - 1)"
    lines.Add "    For i = 1 To results.Count"
    lines.Add "        arr(i - 1) = results(i)"
    lines.Add "    Next i"
    lines.Add "    ReadColumnValues = arr"
    lines.Add "End Function"

    lines.Add "    lb.TopIndex = newIndex"
    lines.Add "End Sub"

    lines.Add ""

        arr(i - 1) = lines(i)
    Next i

End Function
